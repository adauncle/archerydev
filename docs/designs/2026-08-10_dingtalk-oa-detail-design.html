3.1 AuditDriver 抽象基类

# 路径：sql/extensions/audit_drivers/base.py（不在 dingtalk_oa 内）
class AuditDriver(ABC):
    """审批驱动抽象基类。"""
    name: str = ""

    @abstractmethod
    def start(self, workflow, audit, flow) -> DriverStartResult:
        """发起审批。返回 DriverStartResult(external_instance_id, url, raw)。"""
        raise NotImplementedError

    @abstractmethod
    def apply_decision(self, audit, decision: str, operator, remark: str = ""):
        """本地推进后，远程加备注 / 终止流程。"""
        raise NotImplementedError

    @abstractmethod
    def terminate(self, audit, operator, remark: str = ""):
        """远程撤回。"""

    @abstractmethod
    def get_status(self, audit) -> dict:
        """对账用，返回 {status, last_synced_at, raw}。"""

    @abstractmethod
    def handle_callback(self, request) -> HttpResponse:
        """HTTP 回调入口。"""


DRIVER_REGISTRY: Dict[str, Type[AuditDriver]] = {
    "archery": ArcheryDriver,
    "dingtalk_oa": DingtalkOaDriver,
    # 未来扩展：feishu, lark, wecom_oa ...
}

def get_driver(name: str) -> AuditDriver:
    cls = DRIVER_REGISTRY.get(name)
    if not cls:
        raise ValueError(f"unknown audit driver: {name}")
    return cls()

3.2 DingtalkOaDriver

class DingtalkApiError(Exception):
    """钉钉开放平台 API 业务错误（含 errcode != 0）。"""
    def __init__(self, endpoint: str, payload: dict):
        self.endpoint = endpoint
        self.payload = payload
        errcode = payload.get("errcode")
        errmsg = payload.get("errmsg") or payload.get("message")
        super().__init__(f"{endpoint} failed: errcode={errcode} errmsg={errmsg}")


class DingtalkOaDriver(AuditDriver):
    """钉钉 OA 智能工作流 driver。

关键约束:
    * start() 失败按 §8 重试 + 降级，不阻塞业务
      （除非 CUSTOM_DINGTALK_OA_FALLBACK_ENABLED=False）
    * 所有钉钉 API 调用吃掉异常记 logger（不能拖垮本地审批）
    * SQL 全文**不**通过表单传给钉钉，只传摘要 + Archery 详情链接
"""
    name = "dingtalk_oa"

    def start(self, workflow, audit, flow) -> DriverStartResult:
        """发起钉钉 OA 审批。

流程:
    1. 构造脱敏表单（不含 SQL 全文）
    2. 调 topapi.processinstance.create（最多 RETRY_TIMES 次重试）
    3. 成功: 写 WorkflowAuditExternal 关联 + 记 DingtalkOaEventLog(START)
    4. 失败且 FALLBACK_ENABLED=True: 降级到本地 archery flow + 推群告警
    5. 失败且 FALLBACK_ENABLED=False: 抛 DingtalkApiError 阻塞业务
"""
        max_retries = int(getattr(settings, "CUSTOM_DINGTALK_OA_RETRY_TIMES", 3))
        timeout = int(getattr(settings, "CUSTOM_DINGTALK_OA_TIMEOUT_SECONDS", 10))
        fallback = bool(getattr(settings, "CUSTOM_DINGTALK_OA_FALLBACK_ENABLED", True))

        if not flow.dingtalk_process_code:
            # 模板编码缺失是配置错误 → 直接降级（不要死循环重试）
            return self._fallback(audit, flow, reason="process_code 缺失")

        form = self._build_form(workflow, audit, flow)
        last_exc = None
        for attempt in range(1, max_retries + 1):
            try:
                resp = self._call_api(
                    "/topapi/processinstance/create",
                    {"process_code": flow.dingtalk_process_code, "form_component_values": form},
                    timeout=timeout,
                )
                instance_id = resp.get("process_instance_id")
                if not instance_id:
                    raise DingtalkApiError("/topapi/processinstance/create", resp)
                # 写外部关联
                WorkflowAuditExternal.objects.update_or_create(
                    audit=audit,
                    defaults={
                        "driver_name": self.name,
                        "external_instance_id": instance_id,
                        "status": "RUNNING",
                        "external_url": f"https://aflow.dingtalk.com/dingtalk/pc/index.htm?procInsId={instance_id}",
                    },
                )
                DingtalkOaEventLog.objects.create(
                    audit=audit, event_type="START", request_payload=form, response_payload=resp,
                )
                return DriverStartResult(external_instance_id=instance_id, url=..., raw=resp)
            except (DingtalkApiError, requests.RequestException) as exc:
                last_exc = exc
                logger.warning("dingtalk start attempt %s failed: %s", attempt, exc)
                continue
        # 全部重试失败
        if fallback:
            return self._fallback(audit, flow, reason=f"start failed: {last_exc}")
        raise DingtalkApiError("/topapi/processinstance/create", {"errcode": -1, "errmsg": str(last_exc)})

    def _build_form(self, workflow, audit, flow) -> List[dict]:
        """构造脱敏表单（不含 SQL 全文）。"""
        detail_url = f"/sqlworkflow/{workflow.id}/"
        return [
            {"name": "标题", "value": workflow.workflow_name},
            {"name": "实例", "value": workflow.instance.instance_name},
            {"name": "库", "value": workflow.db_name},
            {"name": "SQL 摘要", "value": (workflow.sqlworkflowcontent.sql_content or "")[:100] + "..."},
            {"name": "Archery 详情", "value": detail_url},
            {"name": "发起人", "value": audit.user_display or audit.user_name},
        ]

    def _call_api(self, endpoint: str, payload: dict, timeout: int = 10) -> dict:
        """统一 API 调用（含 access_token 注入 + 错误处理）。

Raises:
    DingtalkApiError: errcode != 0
    requests.RequestException: 网络错误
"""
        access_token = get_oa_access_token()
        url = f"https://oapi.dingtalk.com{endpoint}"
        params = {"access_token": access_token}
        r = requests.post(url, params=params, json=payload, timeout=timeout)
        r.raise_for_status()
        data = r.json()
        if data.get("errcode") != 0:
            raise DingtalkApiError(endpoint, data)
        return data.get("result", data)

    def _fallback(self, audit, flow, reason: str) -> DriverStartResult:
        """降级到本地 archery flow + 推群告警。"""
        audit.audit_driver = "archery"
        audit.save(update_fields=["audit_driver"])
        WorkflowAuditExternal.objects.update_or_create(
            audit=audit, defaults={"driver_name": "archery", "status": "FALLBACK"},
        )
        DingtalkOaEventLog.objects.create(
            audit=audit, event_type="FALLBACK_AT_START", error_message=reason,
        )
        # 推 DBA 群告警（best-effort）
        try:
            send_dingtalk_text(f"⚠️ 钉钉 OA 降级: {audit.id} 原因: {reason}")
        except Exception:
            pass
        # 触发本地流程继续
        return DriverStartResult(external_instance_id="", url="", raw={"fallback": True, "reason": reason})

    def apply_decision(self, audit, decision: str, operator, remark: str = ""):
        """本地推进后，钉钉加备注 / 终止流程。"""
        ext = getattr(audit, "external", None)
        if not ext or ext.driver_name != self.name:
            return
        try:
            if decision in ("passed", "rejected"):
                # 加备注
                self._call_api("/topapi/processinstance/comment/add", {
                    "process_instance_id": ext.external_instance_id,
                    "comment": f"[Archery] {operator} {decision}: {remark}",
                })
                # 终止流程（本地已通过）
                self._call_api("/topapi/processinstance/terminate", {
                    "process_instance_id": ext.external_instance_id,
                    "remark": remark,
                })
            ext.status = "PASSED" if decision == "passed" else "REJECTED"
            ext.last_synced_at = timezone.now()
            ext.save()
        except Exception as exc:
            logger.exception("dingtalk apply_decision failed: %s", exc)

    def terminate(self, audit, operator, remark: str = ""):
        """钉钉侧撤回。"""
        ext = getattr(audit, "external", None)
        if not ext or ext.driver_name != self.name:
            return
        try:
            self._call_api("/topapi/processinstance/terminate", {
                "process_instance_id": ext.external_instance_id,
                "remark": f"[Archery] {operator} 撤回: {remark}",
            })
            ext.status = "TERMINATED"
            ext.last_synced_at = timezone.now()
            ext.save()
            DingtalkOaEventLog.objects.create(
                audit=audit, event_type="TERMINATE", request_payload={"remark": remark},
            )
        except Exception:
            logger.exception("dingtalk terminate failed")

    def get_status(self, audit) -> dict:
        """对账用。"""
        ext = getattr(audit, "external", None)
        if not ext or ext.driver_name != self.name or not ext.external_instance_id:
            return {"status": "UNKNOWN"}
        try:
            data = self._call_api("/topapi/processinstance/get", {
                "process_instance_id": ext.external_instance_id,
            })
            status = data.get("status")  # NEW / RUNNING / COMPLETED / TERMINATED
            # 映射到本系统 status
            ext_map = {"NEW": "PENDING", "RUNNING": "RUNNING",
                      "COMPLETED": "PASSED", "TERMINATED": "TERMINATED"}
            ext.status = ext_map.get(status, "RUNNING")
            ext.last_synced_at = timezone.now()
            ext.save()
            DingtalkOaEventLog.objects.create(
                audit=audit, event_type="SYNC", response_payload=data,
            )
            return {"status": ext.status, "last_synced_at": ext.last_synced_at, "raw": data}
        except Exception as exc:
            logger.warning("dingtalk get_status failed: %s", exc)
            return {"status": "UNKNOWN", "error": str(exc)}

    def handle_callback(self, request):
        """HTTP 回调入口（委托给 services.oa_callback_handler）。"""
        from ..services.oa_callback_handler import handle_oa_callback
        return handle_oa_callback(request, driver=self)

043D 策略匹配算法

4.1 `match_policy`

def match_policy(
    workflow,
    affected_tables: Optional[Sequence[dict]] = None,
) -> Optional[ApprovalPolicy]:
    """根据工单特征匹配一条 ApprovalPolicy。

Args:
    workflow: sql.models.SqlWorkflow 实例
    affected_tables: 可选 [{db, table}, ...]
        传 None 时内部从 workflow.sqlworkflowcontent.sql_content
        调上游 sql.utils.extract_tables.extract_tables 解析

Returns:
    命中的 ApprovalPolicy；未命中返回 None

算法:
    1. 从工单提取三个特征: SQL 类型集合、影响行数、涉及表集合
    2. 按 priority 倒序遍历启用的 ApprovalPolicy
    3. 三维 AND 匹配: SQL 类型 ∩ 业务表 ∩ 影响行数
    4. 第一条命中即返回
    5. 全部不命中 → None（调用方走默认 flow）
"""
    sql_content = workflow.sqlworkflowcontent.sql_content or ""
    sql_types = extract_sql_types(sql_content)
    if affected_tables is None:
        affected_tables = extract_affected_tables(workflow)
    affected_rows = extract_affected_rows(workflow, mode="total")

    policies = (
        ApprovalPolicy.objects.filter(is_enabled=True)
        .select_related("flow")
        .prefetch_related("sql_types")
        .order_by("-priority")
    )
    for policy in policies:
        if not _match_sql_types(policy, sql_types):
            continue
        if policy.require_core_table and not _has_core_table(affected_tables, policy.table_levels):
            continue
        if not _match_affected_rows(policy, affected_rows):
            continue
        return policy
    return None

4.2 三维判定函数

def _match_sql_types(policy: ApprovalPolicy, sql_types: set) -> bool:
    """维度 1: SQL 类型集合。

策略为空集合时永远不命中（防御性）。
match_mode='all' → 全部类型必须有；'any'（默认）→ 任一即可。
"""
    policy_types: set = set(policy.sql_types.values_list("code", flat=True))
    if not policy_types:
        return False
    if policy.sql_type_match_mode == "all":
        return policy_types.issubset(sql_types)
    return bool(policy_types & sql_types)


def _has_core_table(affected_tables: Sequence[dict], levels: str) -> bool:
    """维度 2: 业务表清单中存在受影响的表。

levels 形如 "L1,L2"；空表示任意等级。
命中条件: 任意受影响表 ∈ CoreBusinessTable 且 level ∈ levels。
"""
    if not levels:
        return True
    level_set = set(s.strip() for s in levels.split(",") if s.strip())
    core_dbs = set(CoreBusinessTable.objects.filter(
        level__in=level_set, is_active=True,
    ).values_list("db_name", "table_name"))
    for t in affected_tables:
        if (t["db"], t["table"]) in core_dbs:
            return True
    return False


def _match_affected_rows(policy: ApprovalPolicy, affected_rows: int) -> bool:
    """维度 3: 影响行数在 [min, max] 区间内（None 端不限制）。"""
    if policy.min_affected_rows is not None and affected_rows < policy.min_affected_rows:
        return False
    if policy.max_affected_rows is not None and affected_rows > policy.max_affected_rows:
        return False
    return True

def extract_sql_types(sql_content: str) -> Set[str]:
    """从 SQL 文本提取 SQL 类型集合（去重）。

算法:
    1. sqlparse.parse(sql_content) 分句
    2. 每句 first_token 取大写（如 INSERT/UPDATE/DELETE/DROP/TRUNCATE/CREATE/ALTER/SELECT）
    3. 查 ext_sql_type_registry 匹配（用 SqlTypeRegistry.pattern 正则）
    4. 返回匹配到的 code 集合

Examples:
    >>> extract_sql_types("INSERT INTO t VALUES (1); DROP TABLE t;")
    {"INSERT", "DROP"}

Raises:
    不抛异常（无法识别时返回空集合）
"""
    types = set()
    parsed = sqlparse.parse(sql_content)
    for stmt in parsed:
        if not isinstance(stmt, Statement):
            continue
        first = str(stmt.token_first(skip_cm=True)).upper().strip()
        for reg in SqlTypeRegistry.objects.filter(is_active=True):
            if re.match(reg.pattern, first, re.IGNORECASE):
                types.add(reg.code)
                break
    return types

06回调加密 / 签名

钉钉回调双重保护：签名 = SHA1(sorted([token, timestamp, nonce, encrypted_body]))；加密 = AES-256-CBC，IV = aes_key 前 16 字节；密文 = random(16B) + msg_len(4B) + msg_json + receiveid；填充 PKCS7（block=32）。

6.1 `DingtalkCrypto`

class DingtalkCrypto:
    """钉钉 OA 回调加密/解密 + 签名校验。

Args:
    token: 钉钉后台「事件订阅」生成的 Token 字符串
    aes_key: 43 字符 base64（不含 =），钉钉后台生成
        解码后是 32 字节，作为 AES-256 key + IV（前 16 字节）
    receiveid: 钉钉后台配置的企业 corp_id / receive_id
        用于校验密文尾部、加密回包尾部。空字符串跳过此校验。
"""
    BLOCK_SIZE = 32

    def __init__(self, token: str, aes_key: str, receiveid: str = ""):
        if not token:
            raise ValueError("token is required")
        if not aes_key or len(aes_key) != 43:
            raise ValueError("aes_key must be 43 chars (base64 without padding)")
        self.token = token
        self.receiveid = receiveid or ""
        try:
            self.aes_key = base64.b64decode(aes_key + "=")
        except Exception as e:
            raise ValueError(f"aes_key base64 decode failed: {e}") from e
        if len(self.aes_key) != 32:
            raise ValueError(f"aes_key decoded length must be 32 bytes, got {len(self.aes_key)}")

    def verify_signature(self, timestamp: str, nonce: str, encrypted_b64: str, signature: str) -> bool:
        """校验 URL 参数 signature。

钉钉 v2 签名规则:
    1) 把 token、timestamp、nonce、encrypted_body 排序
    2) 拼接后 SHA1（hex）
    3) 与 URL 参数 signature 比较（hmac.compare_digest 防时序攻击）
"""
        if not signature:
            return False
        params = sorted([self.token, str(timestamp), str(nonce), str(encrypted_b64)])
        expected = hashlib.sha1("".join(params).encode("utf-8")).hexdigest()
        return hmac.compare_digest(expected, str(signature))

    def decrypt(self, encrypted_b64: str) -> dict:
        """AES-256-CBC 解密 + 解析 JSON。

密文布局: random(16B) + msg_len(4B 大端) + msg_json + receiveid

Raises:
    ValueError: 密文长度异常、receiveid 不匹配、msg_len 越界
    json.JSONDecodeError: 解密后非 JSON
"""
        if not encrypted_b64:
            raise ValueError("encrypted body is empty")
        try:
            ciphertext = base64.b64decode(encrypted_b64)
        except Exception as e:
            raise ValueError(f"base64 decode failed: {e}") from e
        if len(ciphertext) < 32:
            raise ValueError(f"ciphertext too short: {len(ciphertext)} bytes (min 32)")

        iv = self.aes_key[:16]
        cipher = AES.new(self.aes_key, AES.MODE_CBC, iv)
        plain = cipher.decrypt(ciphertext)
        plain = plain[16:]  # 跳过前 16 字节 random

        if len(plain) < 4:
            raise ValueError("plaintext header too short")
        msg_len = struct.unpack(">I", plain[:4])[0]
        if msg_len <= 0 or msg_len > len(plain) - 4:
            raise ValueError(f"msg_len out of range: {msg_len}")

        msg = plain[4 : 4 + msg_len]
        if self.receiveid:
            tail = plain[4 + msg_len : 4 + msg_len + len(self.receiveid)]
            expected = self.receiveid.encode("utf-8")
            if tail != expected:
                raise ValueError("receiveid mismatch")
        return json.loads(msg.decode("utf-8"))

    def encrypt(self, msg: Union[dict, str]) -> str:
        """AES-256-CBC 加密回包用。"""
        if isinstance(msg, dict):
            msg_bytes = json.dumps(msg, ensure_ascii=False).encode("utf-8")
        elif isinstance(msg, str):
            msg_bytes = msg.encode("utf-8")
        else:
            raise TypeError(f"msg must be dict or str, got {type(msg).__name__}")

        msg_len = struct.pack(">I", len(msg_bytes))
        random_bytes = os.urandom(16)
        plain = random_bytes + msg_len + msg_bytes + self.receiveid.encode("utf-8")
        # PKCS7 padding
        pad = self.BLOCK_SIZE - len(plain) % self.BLOCK_SIZE
        plain += bytes([pad] * pad)

        iv = self.aes_key[:16]
        cipher = AES.new(self.aes_key, AES.MODE_CBC, iv)
        return base64.b64encode(cipher.encrypt(plain)).decode("utf-8")

# security/guard.py
from django.core.cache import cache

_TOKEN_CACHE_KEY = "dingtalk_oa:access_token"
_TOKEN_TTL = 7000  # 7000s（钉钉 access_token 2h 有效期，提前 100s 刷新）

def get_oa_access_token() -> str:
    """获取企业 access_token（带 cache）。

1. cache 命中 → 直接返回
2. 未命中 → 调 /gettoken 拉新 + 写 cache
3. 过期前自动刷新

Raises:
    RuntimeError: 钉钉返回 errcode != 0
    requests.RequestException: 网络错误
"""
    token = cache.get(_TOKEN_CACHE_KEY)
    if token:
        return token

    app_key = getattr(settings, "CUSTOM_DINGTALK_OA_APP_KEY", "")
    app_secret = getattr(settings, "CUSTOM_DINGTALK_OA_APP_SECRET", "")
    if not app_key or not app_secret:
        raise RuntimeError("CUSTOM_DINGTALK_OA_APP_KEY/SECRET 未配置")

    resp = requests.get(
        "https://oapi.dingtalk.com/gettoken",
        params={"appkey": app_key, "appsecret": app_secret},
        timeout=10,
    )
    data = resp.json()
    if data.get("errcode") != 0:
        raise RuntimeError(f"gettoken failed: {data}")
    token = data["access_token"]
    cache.set(_TOKEN_CACHE_KEY, token, _TOKEN_TTL)
    return token

# services/oa_callback_handler.py
from django.views.decorators.csrf import csrf_exempt
from django.http import HttpResponse, JsonResponse

@csrf_exempt
def handle_oa_callback(request, driver: DingtalkOaDriver):
    """钉钉 OA 回调入口（不含 CSRF，依赖签名校验）。

Args:
    request: Django HttpRequest
    driver: DingtalkOaDriver 实例

Returns:
    JsonResponse / HttpResponse

Raises:
    不抛异常（所有错误写 DingtalkOaEventLog）
"""
    try:
        timestamp = request.GET.get("timestamp", "")
        nonce = request.GET.get("nonce", "")
        signature = request.GET.get("signature", "")
        encrypted = request.body.decode("utf-8")

        crypto = DingtalkCrypto(
            token=getattr(settings, "CUSTOM_DINGTALK_OA_TOKEN", ""),
            aes_key=getattr(settings, "CUSTOM_DINGTALK_OA_AES_KEY", ""),
            receiveid=getattr(settings, "CUSTOM_DINGTALK_OA_RECEIVEID", ""),
        )

        # 1. 验签
        if not crypto.verify_signature(timestamp, nonce, encrypted, signature):
            DingtalkOaEventLog.objects.create(
                event_type="CALLBACK", error_message=f"signature 校验失败: signature={signature}",
            )
            return HttpResponse("signature fail", status=403)

        # 2. 解密
        payload = crypto.decrypt(encrypted)
        DingtalkOaEventLog.objects.create(
            event_type="CALLBACK", request_payload=payload,
        )

        # 3. 路由：bpms_instance_change 等事件
        event_type = payload.get("eventType")
        if event_type in ("bpms_instance_change", "process_instance_change"):
            _handle_instance_change(payload, driver)
        elif event_type == "check_url":
            # 钉钉事件订阅验证（首次配置时）
            return HttpResponse("success")
        else:
            logger.info("unknown event type: %s", event_type)

        # 4. 回包加密（success 字符串）
        encrypted_resp = crypto.encrypt("success")
        return JsonResponse({"msg": "success", "encrypted": encrypted_resp})
    except Exception as exc:
        logger.exception("handle_oa_callback failed: %s", exc)
        DingtalkOaEventLog.objects.create(
            event_type="CALLBACK", error_message=str(exc),
        )
        return HttpResponse(f"error: {exc}", status=500)


def _handle_instance_change(payload: dict, driver: DingtalkOaDriver):
    """bpms_instance_change: 解析 processInstanceId + type + result → 推进本地 audit。"""
    process_instance_id = payload.get("processInstanceId")
    if not process_instance_id:
        return
    try:
        ext = WorkflowAuditExternal.objects.get(external_instance_id=process_instance_id)
    except WorkflowAuditExternal.DoesNotExist:
        logger.warning("unknown processInstanceId: %s", process_instance_id)
        return

    type_ = payload.get("type")  # start / finish
    result = payload.get("result")  # agree / refuse
    operator_userid = payload.get("staffId")

    if type_ == "finish":
        decision = "passed" if result == "agree" else "rejected"
        # 调本地审计推进（复用 Archery 上游 audit handler）
        from common.utils.workflow_audit import Audit
        audit_handler = Audit(audit=ext.audit, operator=_resolve_user(operator_userid))
        if decision == "passed":
            audit_handler.operate(WorkflowAction.PASS, _resolve_user(operator_userid), "钉钉 OA 通过")
        else:
            audit_handler.operate(WorkflowAction.REJECT, _resolve_user(operator_userid), "钉钉 OA 拒绝")
        ext.status = "PASSED" if decision == "passed" else "REJECTED"
        ext.last_synced_at = timezone.now()
        ext.save()
        # 调 driver 加备注（已通过则撤回钉钉流程）
        driver.apply_decision(ext.audit, decision, _resolve_user(operator_userid), "钉钉 OA 自动同步")

08降级 + 重试 + 对账

8.1 重试策略（start）

「灰度开关 + 行为」映射表：DingtalkOaDriver.start() 调钉钉 topapi/processinstance/create 失败时，按下表决定重试或降级。所有重试都在 driver 内部同步进行；超过最大次数后抛 DingtalkApiError，由 §8.2 补偿兜底。

异常分类判定重试次数退避终态
网络超时 / 连接重置 requests.Timeout / ConnectionError / ConnectionResetError 3 指数 1s / 4s / 9s（2^n - 1） 仍失败 → 抛 DingtalkApiError，§8.2 接管 fallback
5xx 服务器错 errcode ∈ {50000, 503, 502, 504} 2 2s / 6s 仍失败 → 抛 DingtalkApiError
429 限流 errcode = 88 (system busy) 3 60s / 180s / 300s 仍失败 → 抛 DingtalkApiError
401 / 403 token 失效 errcode ∈ {88, 40014, 42001, 43004} 0（不重试） — 立即抛 DingtalkApiError("token_invalid")，_force_refresh_token() 刷新一次后再次尝试，失败则 §8.2 接管
400 业务参数错 errcode ∈ {400, 40001, 40002, 40003} 0（不重试） — 立即抛 DingtalkApiError("invalid_param")，_send_admin_alert() 推 DBA 群
业务码：模板未上架 / 已停用 errcode = "processCodeInvalid" 0（不重试） — 立即抛 DingtalkApiError("process_disabled")，推 DBA 群 + 写 DingtalkOaEventLog

8.1.1 重试实现（driver 内部）

# drivers/dingtalk.py
import time
import requests

from django.conf import settings

class DingtalkApiError(Exception):
    """钉钉 OA 业务异常。code: token_invalid / invalid_param / process_disabled / network / unknown"""
    def __init__(self, code: str, message: str, raw: dict | None = None):
        super().__init__(message)
        self.code = code
        self.raw = raw or {}

_RETRY_POLICY = {
    "network":    {"max": 3, "backoff": [1, 4, 9]},
    "5xx":        {"max": 2, "backoff": [2, 6]},
    "429":        {"max": 3, "backoff": [60, 180, 300]},
    "token":      {"max": 0, "backoff": []},
    "invalid":    {"max": 0, "backoff": []},
    "disabled":   {"max": 0, "backoff": []},
}

def _classify_error(exc_or_resp) -> str:
    """把 requests 异常 / 钉钉 errcode 归类为 _RETRY_POLICY 的 key。"""
    if isinstance(exc_or_resp, (requests.Timeout, requests.ConnectionError)):
        return "network"
    errcode = str((exc_or_resp or {}).get("errcode") or "")
    if errcode in ("88", "40014", "42001", "43004"):
        return "token"
    if errcode in ("50000", "503", "502", "504"):
        return "5xx"
    if errcode == "400" or errcode.startswith("4000"):
        return "invalid"
    if errcode in ("processCodeInvalid", "PROCESS_NOT_FOUND"):
        return "disabled"
    return "unknown"

def _retry_with_backoff(func, *args, **kwargs):
    """通用重试包装：根据异常分类查表。"""
    attempt, last_exc = 0, None
    while True:
        try:
            return func(*args, **kwargs)
        except (requests.Timeout, requests.ConnectionError) as e:
            kind = "network"
            last_exc = e
        except DingtalkApiError as e:
            kind = _classify_error(e.raw)
            last_exc = e
        policy = _RETRY_POLICY[kind]
        if attempt >= policy["max"]:
            raise last_exc
        wait = policy["backoff"][attempt]
        logger.warning("dingtalk retry kind=%s attempt=%d wait=%ds", kind, attempt + 1, wait)
        time.sleep(wait)
        attempt += 1

# drivers/dingtalk.py
def _fallback_to_archery(workflow, audit, reason: str):
    """start() 失败时调用：把 driver 切回 archery，写降级原因。"""
    workflow.audit_driver = "archery"

    )

    # 推 DBA 群 + 钉消息给发起人
    _send_admin_alert(f"⚠️ 钉钉 OA 失败降级\n工单 #{audit.audit_id}\n原因：{reason[:200]}")

8.3 对账 cron（reconcile）

钉钉侧的工单状态可能因为回调丢失、token 过期、回调服务重启等原因，停留在 RUNNING 而本地无更新。django-q2 每 5 分钟跑一次对账 task 主动同步：

8.3.1 reconcile_pending_oa_workflows

# tasks.py
from celery import shared_task   # 项目无 Celery，装饰器由 try/except fallback 为 noop
from django_q.models import Schedule
from django_q.tasks import schedule

RECONCILE_SCHEDULE_NAME = "dingtalk_oa_reconcile"

@shared_task
def reconcile_pending_oa_workflows() -> int:
    """每 5 分钟扫一次 RUNNING 中超时的工单。Returns: 本次扫描数（监控用）。"""
    threshold = timezone.now() - timedelta(minutes=settings.CUSTOM_DINGTALK_OA_RECONCILE_TIMEOUT_MIN)
    pending = WorkflowAuditExternal.objects.filter(
        external_status="RUNNING", last_synced_at__lt=threshold,
    ).select_related("audit", "audit__workflow")
    driver = DingtalkOaDriver()
    scanned = 0
    for ext in pending:
        scanned += 1
        try:
            status = driver.get_status(ext.audit) or {}
        except Exception as e:
            ext.reconcile_failed_count = (ext.reconcile_failed_count or 0) + 1
            ext.last_synced_at = timezone.now()
            ext.save(update_fields=["reconcile_failed_count", "last_synced_at"])
            if ext.reconcile_failed_count >= 3:
                _force_fallback(ext, f"reconcile failed 3 times: {e}")
            continue
        # 钉钉 status: COMPLETED/AGREE/REFUSE/TERMINATED
        ext_status = str(status.get("status") or "").upper()
        if ext_status in ("COMPLETED", "APPROVED", "AGREE"):
            ext.external_status, ext.last_synced_at = "DONE", timezone.now()
            ext.save(update_fields=["external_status", "last_synced_at"])
        elif ext_status in ("TERMINATED", "REFUSED", "REJECTED"):
            ext.external_status, ext.last_synced_at = "TERMINATED", timezone.now()
            ext.save(update_fields=["external_status", "last_synced_at"])
        else:
            ext.last_synced_at, ext.reconcile_failed_count = timezone.now(), 0
            ext.save(update_fields=["last_synced_at", "reconcile_failed_count"])
    return scanned

def _force_fallback(ext, reason: str):
    """对账连续 3 次失败：把工单强制降级到本地 Group 审批 + 推 DBA 群。"""
    audit, workflow = ext.audit, ext.audit.get_workflow()
    workflow.audit_driver = "archery"
    workflow.audit_fallback_reason = f"对账失败降级：{reason}"[:255]
    workflow.save(update_fields=["audit_driver", "audit_fallback_reason"])
    ext.external_status, ext.fallback_at, ext.oa_failure_reason = "FALLBACK", timezone.now(), reason[:500]
    ext.save(update_fields=["external_status", "fallback_at", "oa_failure_reason"])
    DingtalkOaEventLog.objects.create(
        audit=audit, event_type="FALLBACK_AT_RECONCILE",
        event_id=f"reconcile-fallback-{audit.audit_id}-{int(timezone.now().timestamp())}",
        payload={"reason": reason}, processed=True, error=reason[:1000],
    )
    DingtalkOaDriver._send_admin_alert(f"⚠️ 钉钉 OA 对账失败，已降级\naudit_id={audit.audit_id}\n原因：{reason[:200]}")

def add_reconcile_schedule():
    """ops 部署后跑一次：注册 django-q2 schedule（每 5 分钟）。"""
    schedule(
        "sql.extensions.dingtalk_oa.tasks.reconcile_pending_oa_workflows",
        name=RECONCILE_SCHEDULE_NAME,
        schedule_type="I",     # I = interval
        minutes=settings.CUSTOM_DINGTALK_OA_RECONCILE_INTERVAL_MIN,
        repeats=-1, timeout=-1,
    )

8.3.3 对账 state machine

/dingtalk_oa/callback + reconcile cron
回调成功 (type=finish, result=agree) PENDING → DONE
回调成功 (type=finish, result=refuse) PENDING → TERMINATED
回调丢失超过 30min 进入对账
对账查到 COMPLETED RUNNING → DONE
对账查到 TERMINATED RUNNING → TERMINATED
对账 get_status 异常 1~2 次 reconcile_failed_count++
对账 get_status 异常 3 次 RUNNING → FALLBACK + 推 DBA 群
start() 失败 1 次 + 灰度开关 关闭 立即 fallback (DRAFT)

9.2 retry_oa 视图实现

# views.py
@permission_required("sql.audit_user", raise_exception=True)
@require_POST
def retry_oa(request, workflow_id: int):
    """手动重试钉钉 OA（仅降级工单显示按钮，v0.7 §10.4.6）。

    流程：1) 重新跑 match_policy 看是否仍命中 dingtalk_oa flow
          2) 调 DingtalkOaDriver.start()
          3) 清空 audit_fallback_reason，driver 切回 dingtalk_oa
    失败时回到原页面 + 错误消息；不抛出。
    """
    workflow = get_object_or_404(SqlWorkflow, pk=workflow_id)
    try:
        affected_tables = extract_affected_tables(workflow)
        policy = match_policy(workflow, affected_tables=affected_tables)
    except Exception as e:
        messages.error(request, f"策略匹配失败：{e}")
        return redirect(reverse("sql:detail", args=[workflow_id]))

    if not policy or policy.flow.audit_driver != "dingtalk_oa":
        messages.error(request, "当前工单策略不要求钉钉 OA，无需重试")
        return redirect(reverse("sql:detail", args=[workflow_id]))

    audit = workflow.get_audit()
    if audit is None:
        messages.error(request, "找不到对应的审批记录")
        return redirect(reverse("sql:detail", args=[workflow_id]))

    try:
        result = DingtalkOaDriver().start(workflow, audit, policy.flow)
    except Exception as e:
        messages.error(request, f"重试失败：{e}")
        return redirect(reverse("sql:detail", args=[workflow_id]))

    if (result.extra or {}).get("fallback"):
        messages.warning(request, f"重试触发降级：{result.extra.get('reason', '')[:200]}")
    else:
        messages.success(request, "已重新发起钉钉 OA")
    return redirect(reverse("sql:detail", args=[workflow_id]))

try:
    from celery import shared_task
except ImportError:
    def shared_task(func):
        """Celery 未装时的 fallback 装饰器（noop + 保留 .delay / .apply_async）。"""
        func.delay = lambda *a, **kw: func(*a, **kw)
        func.apply_async = lambda *a, **kw: func()
        return func
