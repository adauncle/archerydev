01设计原则

走 gh-ost 不用触发器：基于 binlog stream 同步数据，避免触发器性能开销和锁争抢

subprocess.Popen + nohup：不走 systemd-run（systemd 219 缺 --scope），直接 Popen 跑后台

poller 外层 try/except 兜底：daemon thread 任何未捕获异常都写 task.error_message + 切 failed，DBA 可见

进度面板 3s polling：不引 WebSocket/SSE，保持架构简单

        db_table = "ext_ddl_ghost_task"

            models.Index(fields=["status"]),
            models.Index(fields=["created_at"]),

        return f"gh-ost #{self.id} for workflow={self.workflow_id} [{self.status}]"

    def is_terminal(self) -> bool:

        return self.status in ("success", "failed", "cancelled", "rolled_back")

    @property
    def duration_seconds(self) -> int:

        if not self.started_at:
            return 0
        end = self.finished_at or self.updated_at
        return max(0, int((end - self.started_at).total_seconds()))

3.1 主函数

# services/precheck.py
def run_all_prechecks(
    workflow, instance, db_name: str, table_name: str, alter_sql: str
) -> dict:
    """跑 5 道预检 + 收集结果。

Returns:
    {
        "passed": bool,  # 5 道全过
        "summary": str,   # "5/5 PASS" 或 "3/5 PASS (失败: binlog, privs)"
        "checks": [
            {"name": "binlog_format", "passed": True/False, "evidence": ..., "message": ...},
            ...
        ],
        "table_size_bytes": int,  # 用于后续 disk 校验
    }
"""
    checks = []
    # 1. binlog_format = ROW（必需）
    checks.append(check_binlog_format(instance))
    # 2. 磁盘空间 ≥ 1.2 × 表大小
    table_size = _get_table_size(instance, db_name, table_name)
    checks.append(check_disk_space(instance, table_size))
    # 3. REPLICATION SLAVE + CLIENT 权限
    checks.append(check_replication_privileges(instance))
    # 4. ALTER SQL 合法性（不改主键/全文/外键）
    checks.append(check_alter_sql(alter_sql))
    # 5. 表存在 + 是 InnoDB + 非分区
    checks.append(check_table_type(instance, db_name, table_name))

    passed = all(c["passed"] for c in checks)
    failed = [c["name"] for c in checks if not c["passed"]]
    summary = f"{len(checks)-len(failed)}/{len(checks)} PASS" + (
        f" (失败: {', '.join(failed)})" if failed else ""
    )
    return {"passed": passed, "summary": summary, "checks": checks, "table_size_bytes": table_size}

3.2 5 道检查函数

def check_binlog_format(instance) -> dict:
    """binlog_format 必须 = ROW。

gh-ost 基于 binlog stream 同步，STATEMENT/MIXED 不支持。
"""
    try:
        user, password, (host, port) = _get_creds(instance)
        conn = pymysql.connect(host=host, port=port, user=user, password=password, connect_timeout=5)
        try:
            with conn.cursor() as cur:
                cur.execute("SHOW VARIABLES LIKE 'binlog_format'")
                row = cur.fetchone()
                value = row[1] if row else ""
                if value.upper() == "ROW":
                    return {"name": "binlog_format", "passed": True,
                            "evidence": "binlog_format=ROW", "message": "binlog 格式正确"}
                return {"name": "binlog_format", "passed": False,
                        "evidence": f"binlog_format={value}",
                        "message": f"binlog 格式必须是 ROW，当前是 {value}（gh-ost 不支持 STATEMENT/MIXED）"}
        finally:
            conn.close()
    except Exception as exc:
        return {"name": "binlog_format", "passed": False,
                "evidence": "", "message": f"查询失败：{exc}"}


def check_disk_space(instance, table_size_bytes: int) -> dict:
    """磁盘剩余 ≥ 1.2 × 表大小。

gh-ost 创建 ghost table + changelog table + binlog 重放，需要 1.2x 空间。
"""
    try:
        user, password, (host, port) = _get_creds(instance)
        conn = pymysql.connect(host=host, port=port, user=user, password=password, connect_timeout=5)
        try:
            with conn.cursor() as cur:
                cur.execute("SHOW VARIABLES LIKE 'datadir'")
                datadir = cur.fetchone()[1]
                # 用 shutil 查 disk_usage
                import shutil
                usage = shutil.disk_usage(datadir)
                required = int(table_size_bytes * 1.2)
                if usage.free >= required:
                    return {"name": "disk_space", "passed": True,
                            "evidence": f"free={usage.free//1024//1024}MB, need={required//1024//1024}MB",
                            "message": "磁盘空间充足"}
                return {"name": "disk_space", "passed": False,
                        "evidence": f"free={usage.free//1024//1024}MB, need={required//1024//1024}MB",
                        "message": "磁盘空间不足（gh-ost 需要 1.2x 表大小）"}
        finally:
            conn.close()
    except Exception as exc:
        return {"name": "disk_space", "passed": False, "message": f"检查失败：{exc}"}


def check_replication_privileges(instance) -> dict:
    """用户有 REPLICATION SLAVE + CLIENT 权限。

gh-ost 伪装从库拉 binlog，需要这两权限。
"""
    try:
        user, password, (host, port) = _get_creds(instance)
        conn = pymysql.connect(host=host, port=port, user=user, password=password, connect_timeout=5)
        try:
            with conn.cursor() as cur:
                cur.execute(f"SHOW GRANTS FOR '{user}'@'{host}'")
                grants = "\n".join(row[0] for row in cur.fetchall())
                if "REPLICATION SLAVE" in grants.upper() and "REPLICATION CLIENT" in grants.upper():
                    return {"name": "replication_privileges", "passed": True,
                            "message": "REPLICATION SLAVE + CLIENT 权限齐备"}
                missing = []
                if "REPLICATION SLAVE" not in grants.upper():
                    missing.append("REPLICATION SLAVE")
                if "REPLICATION CLIENT" not in grants.upper():
                    missing.append("REPLICATION CLIENT")
                return {"name": "replication_privileges", "passed": False,
                        "message": f"缺少权限：{', '.join(missing)}"}
        finally:
            conn.close()
    except Exception as exc:
        return {"name": "replication_privileges", "passed": False, "message": f"检查失败：{exc}"}


def check_alter_sql(alter_sql: str) -> dict:
    """ALTER SQL 合法性 —— 不改主键/全文索引/外键。

gh-ost 1.1.10 对这些操作有特殊语义，DBA 应直接走传统 ALTER。
"""
    forbidden = [r"\bDROP\s+PRIMARY\s+KEY\b", r"\bDROP\s+FULLTEXT\b", r"\bADD\s+FOREIGN\s+KEY\b", r"\bDROP\s+FOREIGN\s+KEY\b"]
    sql_upper = alter_sql.upper()
    for pattern in forbidden:
        if re.search(pattern, sql_upper, re.IGNORECASE):
            return {"name": "alter_sql", "passed": False,
                    "message": f"检测到 gh-ost 不支持的操作：{pattern}"}
    return {"name": "alter_sql", "passed": True, "message": "ALTER SQL 合法"}


def check_table_type(instance, db_name: str, table_name: str) -> dict:
    """表存在 + ENGINE=InnoDB + 非分区。

gh-ost 不支持 MyISAM / MEMORY / 分区表。
"""
    try:
        user, password, (host, port) = _get_creds(instance)
        conn = pymysql.connect(host=host, port=port, user=user, password=password, connect_timeout=5)
        try:
            with conn.cursor() as cur:
                cur.execute("""
                    SELECT ENGINE, CREATE_OPTIONS
                    FROM INFORMATION_SCHEMA.TABLES
                    WHERE TABLE_SCHEMA=%s AND TABLE_NAME=%s
                """, (db_name, table_name))
                row = cur.fetchone()
                if not row:
                    return {"name": "table_type", "passed": False, "message": f"表 {db_name}.{table_name} 不存在"}
                engine, create_options = row
                if engine.upper() != "INNODB":
                    return {"name": "table_type", "passed": False, "message": f"ENGINE={engine}（gh-ost 只支持 InnoDB）"}
                if create_options and "partitioned" in create_options.lower():
                    return {"name": "table_type", "passed": False, "message": "gh-ost 不支持分区表"}
                return {"name": "table_type", "passed": True, "message": f"ENGINE=InnoDB 非分区"}
        finally:
            conn.close()
    except Exception as exc:
        return {"name": "table_type", "passed": False, "message": f"检查失败：{exc}"}

def _get_creds(instance) -> Tuple[str, str, Tuple[str, int]]:
    """取 instance 的 MySQL 连接凭据（user, password, (host, port)）。

优先级:
    1. CUSTOM_GH_OST_PRECHECK_HOST/PORT/USER/PASSWORD（dev 134 历史 mirage 密文兜底）

"""

            settings.CUSTOM_GH_OST_PRECHECK_USER,
            settings.CUSTOM_GH_OST_PRECHECK_PASSWORD,

    if not user or not password:
        raise ValueError(f"instance #{instance.id} 凭据缺失")
    return (user, password, (instance.host, instance.port))

# services/runner.py
def build_ghost_command(task, instance=None) -> List[str]:
    """根据 task 构建 gh-ost 命令行参数列表（subprocess 用）。

Args:
    task: DdlGhostTask 实例
    instance: SqlWorkflow 对应的 Instance（optional, fallback 用 task.workflow.instance）

Returns:
    list[str] 命令行参数

Raises:
    ValueError: instance 缺失
"""
    inst = instance or (task.workflow.instance if task.workflow_id else None)
    if inst is None:
        raise ValueError("no instance available for gh-ost")

    user, password, (host, port) = _get_creds(inst)
    bin_path = getattr(settings, "CUSTOM_GH_OST_BIN", "/usr/local/bin/gh-ost")

    # ALTER 提取
    alter = task.alter_statement or ""
    # gh-ost 接受完整 SQL 或裸子句
    alter_arg = alter if alter.strip().upper().startswith("ALTER") else f"ALTER TABLE {alter}"

    return [
        bin_path,
        f"--host={host}", f"--port={port}",
        f"--user={user}", f"--password={password}",
        f"--database={task.db_name}", f"--table={task.table_name}",
        f"--alter={alter_arg}",
        "--assume-rbr",              # 134 dev 已知 RBR
        "--allow-on-master",        # 单机 dev 无 replica
        "--execute",
        "--exact-rowcount", "--concurrent-rowcount",
        f"--max-load=Threads_running={task.max_load_threads_running}",
        "--nice-ratio=0",
        "--cut-over=atomic",
        "--initially-drop-ghost-table", "--initially-drop-old-table", "--ok-to-drop-table",
        "--verbose", "--default-retries=120",
    ]

5.2 start_ghost_process

def start_ghost_process(task, instance=None) -> int:
    """启动 gh-ost 子进程（Popen + nohup）。

Returns:
    gh-ost PID（成功）

Raises:
    RuntimeError: 启失败（进程秒退）
"""
    log_dir = _ensure_log_dir()
    log_path = os.path.join(log_dir, f"ghost-{task.id}.log")

    cmd = build_ghost_command(task, instance)
    logger.info("gh-ost start: task_id=%s cmd=%s", task.id, " ".join(shlex.quote(c) for c in cmd))

    # log 文件 chown 到 archery 用户（gunicorn 跑在 archery 用户下）
    try:
        if os.path.exists(log_path):
            os.remove(log_path)
        with open(log_path, "w") as f:
            f.write("")
        try:
            import pwd
            archery_pwd = pwd.getpwnam("archery")
            os.chown(log_path, archery_pwd.pw_uid, archery_pwd.pw_gid)
        except (KeyError, PermissionError, OSError):
            pass
    except PermissionError as exc:
        logger.warning("can't touch log file %s: %s", log_path, exc)

    # Popen 后台跑（start_new_session=True 等价 nohup）
    with open(log_path, "ab", buffering=0) as logf:
        proc = subprocess.Popen(
            cmd,
            stdout=logf, stderr=subprocess.STDOUT, stdin=subprocess.DEVNULL,
            start_new_session=True,  # 等价 nohup
            cwd="/tmp",
        )

    if proc.poll() is not None:
        # 进程秒退，看日志
        with open(log_path, "r", errors="replace") as f:
            err_tail = f.read()[-2000:]
        raise RuntimeError(f"gh-ost 进程秒退 (rc={proc.returncode})。log tail: {err_tail}")

    logger.info("gh-ost started: task_id=%s pid=%s log=%s", task.id, proc.pid, log_path)
    return proc.pid


def stop_ghost_process(pid: int, timeout: int = 10) -> bool:
    """停止 gh-ost 子进程。先 SIGTERM，超时 SIGKILL。

Returns:
    True 成功停止
"""
    if pid is None or pid <= 0:
        return False
    try:
        os.kill(pid, signal.SIGTERM)
    except ProcessLookupError:
        return True  # 进程已经不在了
    except PermissionError:
        logger.error("no permission to SIGTERM pid=%s", pid)
        return False

    import time
    for _ in range(timeout * 2):
        try:
            os.kill(pid, 0)  # probe
        except ProcessLookupError:
            return True
        time.sleep(0.5)

    try:
        os.kill(pid, signal.SIGKILL)
    except ProcessLookupError:
        return True
    return True


def is_alive(pid: int) -> bool:
    """进程是否还活着。"""
    if not pid:
        return False
    try:
        os.kill(pid, 0)
        return True
    except (ProcessLookupError, PermissionError):
        return False


def read_log_tail(log_path: str, max_bytes: int = 65536) -> str:
    """读 log 末尾 max_bytes。"""
    try:
        with open(log_path, "rb") as f:
            f.seek(0, os.SEEK_END)
            size = f.tell()
            f.seek(max(0, size - max_bytes))
            return f.read().decode("utf-8", errors="replace")
    except (FileNotFoundError, PermissionError):
        return ""

# services/parser.py
import re
from dataclasses import dataclass

@dataclass
class GhostParseResult:
    is_done: bool = False           # gh-ost 跑完（done 状态）
    is_failed: bool = False         # gh-ost 异常退出
    pct: int = 0                   # 进度百分比
    rows_copied: int = 0
    rows_total: int = 0
    speed: int = 0                 # 行/秒
    eta_seconds: int = 0
    stage: str = ""
    message: str = ""
    error_message: str = ""

# gh-ost 1.1.10 stdout 行格式正则
_RE_COPY = re.compile(
    r"Copy:\s*(?P\d+)/(?P\d+)\s+(?P\d+)\.\d+%"
)
_RE_DONE = re.compile(r"Done migrating|Copy:\s*\d+/\d+\s+100\.\d+%")
_RE_FAILED = re.compile(r"FATAL|ERROR\s+unable\s+to\s+ALTER")

def parse_ghost_log(log_text: str) -> GhostParseResult:
    """从 gh-ost 末尾 N 行日志解析进度。

关键规则（gh-ost 1.1.10）:
    - `Copy: 12345/67890 18.2%; ...` → 进度行
    - `Time: 11s(total)` → 耗时（括号是字面量，不带空格）
    - `Done migrating ...` → done 状态
    - `FATAL` 行 → failed
"""
    result = GhostParseResult()
    lines = log_text.splitlines()

    # 取最后一行 Copy: 进度
    for line in reversed(lines):
        m = _RE_COPY.search(line)
        if m:
            result.rows_copied = int(m.group("copied"))
            result.rows_total = int(m.group("total"))
            result.pct = int(m.group("pct"))
            break

    # 检查 done / failed
    for line in reversed(lines):
        if _RE_FAILED.search(line):
            result.is_failed = True
            result.error_message = line.strip()
            break
        if _RE_DONE.search(line):
            result.is_done = True
            break

    # 最后一行作为 message
    if lines:
        result.message = lines[-1].strip()[:500]

    return result

踩坑：gh-ost 1.1.10 Time: 11s(total) 括号是字面量，不是空格。正则必须 Time:\s*(\d+)s 单独匹配，不要写 Time:\s*(\d+)\s*\(total\)。

07poller：3s 轮询 + 暂停阈值

7.1 入口 + 主循环

# services/poller.py
import threading

POLL_INTERVAL = 3  # 秒
THREADS_RUNNING_QUERY_INTERVAL = 10  # 10s 查一次 Threads_running（不要每 3s 都查）

def start_poller(task_id: int) -> threading.Thread:
    """启 daemon thread 跑 poll_loop。"""
    t = threading.Thread(target=poll_loop, args=(task_id,), daemon=True)
    t.start()
    return t


def poll_loop(task_id: int):
    """主循环：每 3s 读 log + 解析 + 写 progress + 状态机。

终态时自动 return。**外层 try/except 兜底**：任何未捕获异常都写到 task.error_message，
并把 task 切到 failed 状态，让 DBA 能看到。
"""
    try:
        task = DdlGhostTask.objects.get(pk=task_id)
    except DdlGhostTask.DoesNotExist:
        logger.error("poll_loop: task %s not found", task_id)
        return

    log_dir = getattr(settings, "CUSTOM_GH_OST_LOG_DIR", "/var/log/archery/gh_ost")
    log_path = os.path.join(log_dir, f"ghost-{task_id}.log")
    try:
        instance = task.workflow.instance if task.workflow_id else None
    except Exception:
        instance = None
    last_threads_check = 0

    # 标记 running
    if task.status == "queued":
        task.status = "running"
        task.current_stage = "connecting"
        task.started_at = task.started_at or timezone.now()
        task.save()

    try:
        while True:
            # 进程死了？
            try:
                alive = is_alive(task.ghost_pid) if task.ghost_pid else False
            except Exception:
                logger.exception("is_alive failed")
                alive = False

            if not alive:
                # 进程退出，parse log 看 done/failed
                try:
                    tail = read_log_tail(log_path, max_bytes=8192)
                    result = parse_ghost_log(tail)
                except Exception as exc:
                    logger.exception("read/parse log failed on dead process")
                    _finalize_task(task, "failed", f"log 读取失败：{exc}")
                    break
                if result.is_failed:
                    _finalize_task(task, "failed", result.error_message or "gh-ost 进程异常退出")
                elif result.is_done:
                    _finalize_task(task, "success", "")
                else:
                    _finalize_task(task, "failed", "gh-ost 进程退出但未标记成功")
                break

            # 进程还活着：读 log + 解析
            try:
                tail = read_log_tail(log_path, max_bytes=16384)
                result = parse_ghost_log(tail)
            except Exception as exc:
                logger.warning("parse log failed: %s", exc)
                time.sleep(POLL_INTERVAL)
                continue

            # 写 progress 快照
            try:
                task.progress_pct = result.pct
                task.progress_rows_copied = result.rows_copied
                task.progress_rows_total = result.rows_total
                task.progress_speed_rows_per_sec = result.speed
                task.progress_eta_seconds = result.eta_seconds
                task.progress_message = result.message
                task.current_stage = result.stage
                task.last_heartbeat_at = timezone.now()
                task.save(update_fields=[
                    "progress_pct", "progress_rows_copied", "progress_rows_total",
                    "progress_speed_rows_per_sec", "progress_eta_seconds",
                    "progress_message", "current_stage", "last_heartbeat_at",
                ])
            except Exception as exc:
                logger.warning("task.save failed in poll_loop: %s", exc)

            # 解析到 done / failed 立即结束
            if result.is_done:
                _finalize_task(task, "success", "")
                break
            if result.is_failed:
                _finalize_task(task, "failed", result.error_message or "")
                break

            # 暂停阈值（Threads_running 超过 task.max_load_threads_running → SIGSTOP）
            if instance and time.time() - last_threads_check > THREADS_RUNNING_QUERY_INTERVAL:
                _maybe_pause_resume(task, instance)
                last_threads_check = time.time()

            time.sleep(POLL_INTERVAL)
    except Exception as exc:
        # 任何未捕获异常 — 写 task 错误 + 切 failed
        logger.exception("poll_loop uncaught exception")
        try:
            task.error_message = f"poller 异常：{exc}"
            task.status = "failed"
            task.finished_at = timezone.now()
            task.save()
        except Exception:
            logger.exception("task.save failed in fallback")

7.2 终态化 + 暂停阈值

def _finalize_task(task, new_status: str, error_message: str = ""):
    """终态收尾：写状态、停止进程、钉钉通知。"""
    task.status = new_status
    task.finished_at = timezone.now()
    if error_message:
        task.error_message = (task.error_message or "") + "\n" + error_message
    if task.ghost_pid:
        try:
            os.kill(task.ghost_pid, signal.SIGTERM)
        except (ProcessLookupError, PermissionError):
            pass
    task.save()
    # 钉钉通知（best-effort）
    try:
        notify_terminal(task)
    except Exception:
        logger.exception("notify_terminal failed: task=%s", task.id)


def _query_threads_running(instance) -> Optional[int]:
    """查 MySQL Threads_running 状态值。"""
    try:
        user, password, (host, port) = _get_creds(instance)
        conn = pymysql.connect(host=host, port=port, user=user, password=password, connect_timeout=5)
        try:
            with conn.cursor() as cur:
                cur.execute("SHOW GLOBAL STATUS LIKE 'Threads_running'")
                row = cur.fetchone()
                return int(row[1]) if row else None
        finally:
            conn.close()
    except Exception as exc:
        logger.debug("query Threads_running failed: %s", exc)
        return None


def _maybe_pause_resume(task, instance):
    """根据 Threads_running 决定 SIGSTOP / SIGCONT gh-ost 进程。"""
    if not task.ghost_pid or not is_alive(task.ghost_pid):
        return
    threads = _query_threads_running(instance)
    if threads is None:
        return
    if threads > task.max_load_threads_running:
        try:
            os.kill(task.ghost_pid, signal.SIGSTOP)
        except (ProcessLookupError, PermissionError):
            pass
    else:
        try:
            os.kill(task.ghost_pid, signal.SIGCONT)
        except (ProcessLookupError, PermissionError):
            pass

外层 try/except 兜底是必须的：daemon thread 任何未捕获异常会静默死掉，DBA 看不到。poller 失败要把 task.error_message 写好 + 切 failed。

import requests
from django.conf import settings

def notify_terminal(task) -> None:
    """任务终态时发钉钉群通知（best-effort）。

触发:
    - 终态：success / failed / cancelled / rolled_back
    - 灰度开关 DINGTALK_NOTIFY_WEBHOOK 非空
    - best-effort：失败不抛错，只 logger.warning
"""
    webhook = getattr(settings, "DINGTALK_NOTIFY_WEBHOOK", "")
    if not webhook:
        logger.debug("DINGTALK_NOTIFY_WEBHOOK 未配置，skip 钉钉通知")
        return

    status_emoji = {
        "success": "✅", "failed": "❌",
        "cancelled": "⏹", "rolled_back": "↩️",
    }.get(task.status, "❓")

    duration = task.duration_seconds
    if task.workflow_id:
        workflow_name = task.workflow.workflow_name
        url = f"/gh_ost/progress/{task.workflow_id}/"
    else:
        workflow_name = task.target_table or f"{task.db_name}.{task.table_name}"
        url = ""

- 工单/目标: {workflow_name}
- 实例: {task.workflow.instance.instance_name if task.workflow_id else ""}

- 发起人: {task.created_by}
- 耗时: {duration}s"""
    if task.error_message:
        msg += f"\n- 错误: {task.error_message[:500]}"

    if url:
        msg += f"\n[查看进度]({url})"

    payload = {"msgtype": "markdown", "markdown": {"title": "gh-ost 任务完成", "text": msg}}
    try:
        requests.post(webhook, json=payload, timeout=5)
    except Exception as exc:
        logger.warning("dingtalk notify failed: %s", exc)

def enable(request: HttpRequest, workflow_id: int) -> JsonResponse:
    """启用 gh-ost：预检通过后写 DdlGhostTask（status=queued）。"""
    workflow = get_object_or_404(SqlWorkflow, pk=workflow_id)

    parsed = _parse_first_alter(_workflow_sql_text(workflow))
    if not parsed:
        return JsonResponse({"ok": False, "error": "未找到 ALTER TABLE 语句"}, status=400)

    db_name = parsed["db"] or workflow.db_name
    table_name = parsed["table"]

    # 已经存在 task？
    existing = DdlGhostTask.objects.filter(workflow=workflow).first()
    if existing and existing.status in ("running", "cut_over", "queued"):
        return JsonResponse({"ok": False, "error": f"task 已在执行中（status={existing.status}）"}, status=409)

    instance = workflow.instance
    report = run_all_prechecks(workflow=workflow, instance=instance, db_name=db_name, table_name=table_name, alter_sql=parsed["full"])

    if not report["passed"]:
        # 仍然写一条 task 记录（precheck_failed），便于审计
        task = _upsert_task(workflow, parsed, db_name, table_name, passed=False, report=report, created_by=request.user.username)
        return JsonResponse({"ok": False, "passed": False, "summary": report["summary"], "checks": report["checks"], "task_id": task.id}, status=422)

    # 预检通过 → 写 task（status=queued）
    task = _upsert_task(workflow, parsed, db_name, table_name, passed=True, report=report, created_by=request.user.username)
    return JsonResponse({"ok": True, "passed": True, "summary": report["summary"], "task_id": task.id, "status": task.status})

# views.py
@login_required
@require_POST
def start(request: HttpRequest, workflow_id: int) -> JsonResponse:
    """启 gh-ost 子进程 + 启动后台 poller。

流程:
    1. 拿 task，状态校验
    2. runner.start_ghost_process Popen gh-ost
    3. 写 PID + started_at
    4. 启动 poller.start_poller daemon thread（3s 轮询）
"""
    task = get_object_or_404(DdlGhostTask, workflow_id=workflow_id)
    if task.status != "queued":
        return JsonResponse({"ok": False, "error": f"当前状态 {task.status}，不能启动（需 queued）"}, status=409)
    if not task.precheck_passed:
        return JsonResponse({"ok": False, "error": "预检未通过，不能启动"}, status=409)

    instance = task.workflow.instance if task.workflow_id else None
    if instance is None:
        return JsonResponse({"ok": False, "error": "工单没有关联实例"}, status=400)

    try:
        pid = start_ghost_process(task, instance=instance)
    except Exception as exc:
        logger.exception("start_ghost_process failed: task=%s", task.id)
        task.status = "failed"
        task.error_message = f"启动 gh-ost 失败：{exc}"
        task.finished_at = timezone.now()
        task.save()
        return JsonResponse({"ok": False, "error": f"启动 gh-ost 失败：{exc}"}, status=500)

    task.ghost_pid = pid
    task.status = "running"
    task.started_at = timezone.now()
    task.current_stage = "connecting"
    task.progress_pct = 0
    task.progress_message = "gh-ost 已启动，等待连接"
    task.last_heartbeat_at = timezone.now()
    task.save()

    try:
        start_poller(task.id)
    except Exception:
        logger.exception("start_poller failed for task %s", task.id)
        task.error_message = "poller 启失败 — gh-ost 在跑但没人在轮询，请 DBA 介入"
        task.save()

    return JsonResponse({"ok": True, "status": task.status, "task_id": task.id, "pid": pid})

# views.py
@login_required
@require_POST
def rollback(request: HttpRequest, workflow_id: int) -> JsonResponse:
    """DBA 手动回滚：drop 影子表 + 标 rolled_back。

注意：cut-over 成功后影子表 __gho 已 drop，rollback 仅能标 rolled_back。
cut-over 失败/cancelled 状态可 drop _gho/_del 残留。
"""
    task = DdlGhostTask.objects.filter(workflow_id=workflow_id).first()
    if not task:
        return JsonResponse({"ok": False, "error": "task 不存在"}, status=404)
    if task.status not in ("success", "failed", "cancelled"):
        return JsonResponse({"ok": False, "error": f"当前状态 {task.status}，只能在终态后回滚"}, status=409)

    instance = task.workflow.instance
    dropped = []
    errors = []
    if instance:
        try:
            user, password, (host, port) = _get_creds(instance)
            conn = pymysql.connect(host=host, port=port, user=user, password=password, database=task.db_name, connect_timeout=5, autocommit=True)
            try:
                with conn.cursor() as cur:
                    for tbl in [task.ghost_table_name, f"_{task.table_name}_del"]:
                        if tbl:
                            try:
                                cur.execute(f"DROP TABLE IF EXISTS `{task.db_name}`.`{tbl}`")
                                dropped.append(tbl)
                            exceptExceptionas exc:
                                errors.append(f"{tbl}: {exc}")
            finally:
                conn.close()
        exceptExceptionas exc:
            errors.append(f"connect: {exc}")

    task.status = "rolled_back"
    task.finished_at = timezone.now()
    task.error_message = (task.error_message or"") + f"\n[rollback] dropped={dropped} errors={errors}"
    task.save()
    return JsonResponse({"ok": True, "status": task.status, "dropped": dropped, "errors": errors})
